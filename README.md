🚀 Getting Started
Prerequisites
Node.js (v18 or higher)

npm (v9 or higher)

Ionic CLI

Installation
bash
# Clone the repository
git clone https://github.com/thokozani828/SHESHA-JOBELA-Digital.git

# Navigate to the project directory
cd SHESHA-JOBELA-Digital

# Install dependencies
npm install --legacy-peer-deps

# Start the development server
npm start

# Build for production
npm run build
Deployment
bash
# Deploy to GitHub Pages
npm run deploy

# Or build and deploy manually
ng build --configuration production --base-href /SHESHA-JOBELA-Digital/
npx angular-cli-ghpages --dir=www
